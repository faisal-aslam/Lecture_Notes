% ============================================================
% Shared preamble for the Lattice-Based PQC lecture series.
% Included by every lecture file via \input{lecture-preamble}
% Keeps formatting (colors, boxes, title style) consistent
% across all lectures without duplicating code.
% ============================================================

\usepackage[margin=1in]{geometry}
\usepackage{times}
\usepackage{amsmath,amssymb,amsthm}
% algorithm/algpseudocode must load *before* hyperref: algorithm.sty
% pulls in the `float` package internally (via \newfloat), and
% float+hyperref have a well-known load-order incompatibility that
% causes hyperref to register a duplicate PDF destination for every
% single caption in the document (not just algorithms) if hyperref
% loads first. Loading order here is deliberate -- don't reorder.
\usepackage{algorithm}
\usepackage{algpseudocode}
\usepackage{hyperref}
\usepackage{xr-hyper}
% xr-hyper lets \ref{} pull a number from another lecture's .aux file.
% Each lecture file declares which earlier lectures it cross-references
% via \externaldocument{<basename>} right after \input{lecture-preamble}.
% This requires that lecture's .tex to have been compiled at least once
% (compile_all.sh always does lectures in order, so this "just works"
% for a full build; compiling a single later lecture in isolation before
% its dependencies exist will show "??" until you compile the earlier
% one first).
\usepackage{enumitem}
\usepackage{titlesec}
\usepackage{parskip}
\usepackage{tikz}
\usetikzlibrary{arrows.meta,positioning,shapes.geometric}
\usepackage[most]{tcolorbox}
\usepackage{xcolor}

\hypersetup{colorlinks=true, linkcolor=blue, citecolor=blue, urlcolor=blue}
\titlespacing*{\section}{0pt}{14pt}{6pt}
\titlespacing*{\subsection}{0pt}{10pt}{4pt}

% --- Colored boxes for Definitions / Examples / Remarks / Exercises ---
\newtcolorbox{defbox}[1]{colback=blue!5!white, colframe=blue!45!black, fonttitle=\bfseries, title={Definition: #1}, boxrule=0.6pt, arc=2pt}
\newtcolorbox{exbox}[1]{colback=green!5!white, colframe=green!30!black, fonttitle=\bfseries, title={Example: #1}, boxrule=0.6pt, arc=2pt}
\newtcolorbox{rmkbox}[1]{colback=yellow!8!white, colframe=orange!60!black, fonttitle=\bfseries, title={Remark: #1}, boxrule=0.6pt, arc=2pt}
\newtcolorbox{exercisebox}[1]{colback=red!4!white, colframe=red!45!black, fonttitle=\bfseries, title={Exercise: #1}, boxrule=0.6pt, arc=2pt}
\newtcolorbox{previewbox}[1]{colback=purple!5!white, colframe=purple!45!black, fonttitle=\bfseries, title={Looking Ahead: #1}, boxrule=0.6pt, arc=2pt}
\newtcolorbox{notebox}{colback=gray!6!white, colframe=gray!50!black, boxrule=0.5pt, arc=2pt}
\newtcolorbox{factbox}[1]{colback=teal!5!white, colframe=teal!50!black, fonttitle=\bfseries, title={Fact: #1}, boxrule=0.6pt, arc=2pt}
